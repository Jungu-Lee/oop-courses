점진적 타이핑 (gradual typing)

설계
각각을 정확하게 적는 어휘가 필요하다. 아래 코드를 읽으며 힌트 하나하나가 무엇을 약속하는지 찾아본다.
vocab.py
import math
from collections.abc import Callable, Iterable, Sequence
from typing import Literal

Vector3 = tuple[float, float, float]      # 별칭: 긴 타입에 이름을 붙인다
Reading = tuple[float, float]             # (측정 시각, 측정값)


def norm(v: Vector3) -> float:
    return math.sqrt(v[0] ** 2 + v[1] ** 2 + v[2] ** 2)


def mean(values: Sequence[float]) -> float:
    return sum(values) / len(values)


def parse_line(line: str) -> Reading | None:
    parts = line.split(",")
    if len(parts) != 2:
        return None                       # 형식이 틀린 줄은 None
    return float(parts[0]), float(parts[1])


def apply(fn: Callable[[float], float], values: Iterable[float]) -> list[float]:
    return [fn(v) for v in values]


def open_log(mode: Literal["r", "w"]) -> str:
    return f"로그를 '{mode}' 모드로 연다"


def half(x: float) -> float:
    return x / 2


if __name__ == "__main__":
    print(norm((3.0, 4.0, 0.0)))
    print(mean([1, 2, 3]))
    print(parse_line("0.5,21.3"), parse_line("깨진 줄"))
    print(apply(half, [1.0, 2.5]))
    print(open_log("r"))

from typing import reveal_type

calibration: dict[str, float] = {"temp": 0.98, "press": 1.02, "offset": 0.0}


def corrected(sensor: str, raw: float) -> float:
    factor = calibration.get(sensor)
    reveal_type(factor)                   # ①
    if factor is None:                    # None 인 경우를 먼저 처리하고 벗어난다
        raise KeyError(f"보정 계수가 없는 센서: {sensor}")
    reveal_type(factor)                   # ②
    return raw * factor

상황

설계
원소 타입을 매개변수 T 로 둔 제네릭 클래스 Stack[T] 를 만들고, 사용하는 쪽에서 Stack[str] · Stack[float] 로 채운다.
stack.py
class Stack[T]:
    def __init__(self) -> None:
        self._items: list[T] = []

    def push(self, item: T) -> None:
        self._items.append(item)

    def pop(self) -> T:
        if not self._items:
            raise IndexError("빈 스택에서 pop 했습니다")
        return self._items.pop()

    def peek(self) -> T | None:           # 비어 있으면 꺼낼 값이 없다
        return self._items[-1] if self._items else None

    def __len__(self) -> int:
        return len(self._items)

peek() 는 비어 있을 때 꺼낼 값이 없으므로 T | None 을 돌려주고, pop() 은 빈 스택에서 예외를 낸다. 같은 ‘비어 있음’을 두 가지 방식으로 다룬 것은 의도된 설계다 — 값이 없는 것이 정상일 때는 None, 호출자의 실수일 때는 예외.

요구사항

  • add(key, value) — 키와 값을 저장한다.
  • get(key) — 값 또는 None 을 돌려준다 (없을 수 있다).
  • find(predicate) — 조건 함수가 True 인 값들의 목록을 돌려준다. 조건 함수는 값 타입을 받는다.

성공 기준

  1. Repository[K, V] 를 작성한다 (빈칸 없음 — 설계부터 직접 한다).
  2. mypy --strict 결과가 정확히 2개의 오류여야 한다 — ① 학번으로 실험 조회, ② 학생 저장소에 실험 추가. 그 밖의 오류는 없어야 한다.
  3. find(lambda s: s.grade == 2) 에서 s 의 타입이 Student 로 추론되어야 한다.

class Repository[K, V]:
    def __init__(self) -> None:
        self._data: dict[K, V] = {}

    def add(self, key: K, value: V) -> None:
        self._data[key] = value

    def get(self, key: K) -> V | None:
        return self._data.get(key)

    def find(self, predicate: Callable[[V], bool]) -> list[V]:
        return [v for v in self._data.values() if predicate(v)]
